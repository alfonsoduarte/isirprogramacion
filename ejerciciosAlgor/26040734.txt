leer b
leer c
leer d

si a < b y a < c y a < d Entonces
	m <- a
SiNo
	si b < c y b < d Entonces
		m <- b
	SiNo
		si c < d Entonces

		SiNo
			m <- d
		FinSi
	FinSi
FinSi

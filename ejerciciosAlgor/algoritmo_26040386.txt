algoritmo promedio_tres_calificaciones

variables
   c1, c2, c3, promedio : real

inicio

   leer c1

   imprimir "Ingrese la calificacion 2 = "
   leer c2

   imprimir "Ingrese la calificacion 3 = "
   leer c3

   imprimirln "El promedio es: ", promedio
